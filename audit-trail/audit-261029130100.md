## 1. Ringkasan Eksekutif

Sistem lisensi dari `C:\laragon\www\pertokenan` dirancang berbasis **Google Apps Script Web App** dan **Google Sheets** multi-aplikasi. Modul referensi pada `pertokenan/license.md` menggunakan ekosistem PHP. Untuk proyek **WilkerStat Checker (Python / Flask)**, seluruh mekanisme diadaptasi ke dalam modul Python native tanpa dependensi eksternal berat, mencakup:
1. **Hardware ID (HWID) Binding**: Ekstraksi identitas CPU, Motherboard, dan UUID melalui PowerShell `Get-CimInstance` di-hash SHA-256.
2. **Auto-Follow HTTP 302**: `urllib.request` native yang menangani redirect Google Apps Script ke `googleusercontent.com`.
3. **Smart Cache 24 Jam**: Eliminasi latensi jaringan (0 ms) pada operasi harian, disimpan di `data/license_cache.json`.
4. **Flask Sentinel Middleware**: Intersepsi `before_request` pada seluruh endpoint kecuali rute aktivasi dan berkas statis vendor.
5. **Fail-Closed Protection**: Jika berkas lisensi dihapus atau dirusak, aplikasi seketika berhenti (*halt*).

## 3. Komponen Klien Python (`license_manager.py`)

Modul mandiri baru ditempatkan di root proyek: `c:\laragon\www\wilkerstat-checker\license_manager.py`.

```python
"""
WilkerStat Checker — License Manager Client
Integrasi Central License Gateway (Google Apps Script) + HWID Lock + 24-Hour Cache.
"""
import os
import sys
import json
import time
import hashlib
import subprocess
import urllib.request
import urllib.parse
from datetime import datetime
from typing import Dict, Any, Tuple

import config

DEFAULT_API_URL = "https://script.google.com/macros/s/AKfycby0vtuSERXWlNXMnxIPspHM1K7tCxXmKkbPnmXxqJnvMokLxB9uMDPMLSdqobG1cpKt/exec"
CACHE_FILE = os.path.join(config.BASE_DIR, "data", "license_cache.json")
_CACHED_HWID = None


def get_hardware_id() -> str:
    """Membaca identitas perangkat keras unik (CPU + Motherboard UUID)."""
    global _CACHED_HWID
    if _CACHED_HWID is not None:
        return _CACHED_HWID

    raw = ""
    if sys.platform == "win32":
        try:
            cmd = (
                'powershell -NoProfile -Command "'
                '(Get-CimInstance Win32_Processor).ProcessorId + \'|\' + '
                '(Get-CimInstance Win32_BaseBoard).SerialNumber + \'|\' + '
                '(Get-CimInstance Win32_ComputerSystemProduct).UUID"'
            )
            out = subprocess.check_output(cmd, shell=True, text=True, stderr=subprocess.DEVNULL)
            raw = out.strip()
        except Exception:
            pass

    if not raw or raw.strip("|") == "":
        import socket
        import platform
        raw = f"{socket.gethostname()}|{platform.machine()}|{platform.processor()}"

    _CACHED_HWID = hashlib.sha256(raw.encode("utf-8")).hexdigest().upper()
    return _CACHED_HWID


def verify_license() -> Dict[str, Any]:
    """Verifikasi lisensi lokal (cache) dan sinkronisasi ke server gateway."""
    api_url = (getattr(config, "LICENSE_API_URL", "") or DEFAULT_API_URL).strip()
    satker = getattr(config, "LICENSE_SATKER", "").strip()
    token = getattr(config, "LICENSE_TOKEN", "").strip()

    if not api_url:
        return {"valid": False, "message": "LICENSE_API_URL belum dikonfigurasi di file .env."}
    if not satker or not token:
        return {"valid": False, "message": "Kode Satker atau Token Lisensi belum diisi."}

    # 1. Cek Cache Lokal 24 Jam (0 ms Latency)
    if os.path.isfile(CACHE_FILE):
        try:
            with open(CACHE_FILE, "r", encoding="utf-8") as f:
                cached = json.load(f)
            if (
                isinstance(cached, dict)
                and cached.get("valid") is True
                and cached.get("satker") == satker
                and cached.get("token") == token
                and (time.time() - cached.get("cached_at", 0) < 86400)
            ):
                exp_str = cached.get("expires")
                if exp_str:
                    exp_date = datetime.strptime(exp_str, "%Y-%m-%d").date()
                    if datetime.now().date() > exp_date:
                        pass  # Paksa re-check ke API
                    else:
                        return cached
                else:
                    return cached
        except Exception:
            pass

    # 2. Verifikasi Online ke Server Gateway
    hwid = get_hardware_id()
    params = urllib.parse.urlencode({
        "app": "wilkerstat",
        "satker": satker,
        "token": token,
        "hwid": hwid
    })
    sep = "&" if "?" in api_url else "?"
    req_url = f"{api_url}{sep}{params}"

    try:
        req = urllib.request.Request(
            req_url,
            headers={"User-Agent": "Mozilla/5.0 WilkerStatChecker-Client"}
        )
        with urllib.request.urlopen(req, timeout=15) as resp:
            body = resp.read().decode("utf-8")
            data = json.loads(body)

        if not isinstance(data, dict) or "valid" not in data:
            return {"valid": False, "message": "Format respon gateway lisensi tidak valid."}

        if data.get("valid") is True:
            data["cached_at"] = time.time()
            data["satker"] = satker
            data["token"] = token
            data["hwid"] = hwid
            os.makedirs(os.path.dirname(CACHE_FILE), exist_ok=True)
            with open(CACHE_FILE, "w", encoding="utf-8") as f:
                json.dump(data, f)
        else:
            if os.path.isfile(CACHE_FILE):
                try:
                    os.remove(CACHE_FILE)
                except Exception:
                    pass

        return data

    except Exception as e:
        # Grace period offline 3 hari jika server Google gangguan sementara
        if os.path.isfile(CACHE_FILE):
            try:
                with open(CACHE_FILE, "r", encoding="utf-8") as f:
                    cached = json.load(f)
                if cached.get("valid") is True and (time.time() - cached.get("cached_at", 0) < 86400 * 3):
                    return cached
            except Exception:
                pass
        return {"valid": False, "message": f"Gagal menghubungi server lisensi ({e}). Pastikan koneksi internet aktif."}


def save_license_credentials(satker: str, token: str) -> None:
    """Simpan Satker & Token baru ke .env dan reset cache lisensi."""
    env_path = os.path.join(config.BASE_DIR, ".env")
    lines = []
    if os.path.isfile(env_path):
        with open(env_path, "r", encoding="utf-8") as f:
            lines = f.readlines()

    keys = {"LICENSE_SATKER": satker.strip(), "LICENSE_TOKEN": token.strip()}
    found_keys = set()
    new_lines = []

    for line in lines:
        matched = False
        for k, v in keys.items():
            if line.strip().startswith(f"{k}=") or line.strip().startswith(f"#{k}="):
                new_lines.append(f"{k}={v}\n")
                found_keys.add(k)
                matched = True
                break
        if not matched:
            new_lines.append(line)

    for k, v in keys.items():
        if k not in found_keys:
            new_lines.append(f"{k}={v}\n")

    with open(env_path, "w", encoding="utf-8") as f:
        f.writelines(new_lines)

    if os.path.isfile(CACHE_FILE):
        try:
            os.remove(CACHE_FILE)
        except Exception:
            pass
```

## 5. Integrasi Sentinel Middleware pada `app.py`

Tambahkan hook `before_request` di dalam fungsi `create_app()` pada [app.py](file:///c:/laragon/www/wilkerstat-checker/app.py):

```python
from license_manager import verify_license, save_license_credentials, get_hardware_id

@app.before_request
def license_sentinel():
    # Bypass file statis dan endpoint aktivasi lisensi
    if request.path.startswith("/static") or request.path.startswith("/license"):
        return None

    status = verify_license()
    if not status.get("valid"):
        return render_template(
            "license_locked.html",
            message=status.get("message", "Akses aplikasi terkunci."),
            satker=getattr(config, "LICENSE_SATKER", ""),
            token=getattr(config, "LICENSE_TOKEN", ""),
            hwid=get_hardware_id()
        ), 403

@app.route("/license/activate", methods=["POST"])
def activate_license():
    satker = request.form.get("satker", "").strip()
    token = request.form.get("token", "").strip()
    if not satker or not token:
        flash("Kode Satker dan Token wajib diisi.", "danger")
        return redirect(url_for("index"))

    save_license_credentials(satker, token)
    # Reload modul config
    import importlib
    importlib.reload(config)
    return redirect(url_for("index"))
```
